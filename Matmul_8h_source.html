28template <typename T>
29[[nodiscard]] Tensor<T> matmul(const Tensor<T>& a, const Tensor<T>& b) {
30 PROJECTGAMING_ASSERT(a.rank() == 2 && b.rank() == 2,
31 "matmul() : les deux tenseurs doivent etre de rang 2");
32 PROJECTGAMING_ASSERT(
33 a.shape()[1] == b.shape()[0],
34 "matmul() : dimension interne incompatible (colonnes de a != lignes de b)");
35
36 const std::size_t m = a.shape()[0];
37 const std::size_t k = a.shape()[1];
38 const std::size_t n = b.shape()[1];
39
40 const T* aData = a.data();
41 const T* bData = b.data();
42 const std::size_t aRowStride = a.strides()[0];
43 const std::size_t aColStride = a.strides()[1];
44 const std::size_t bRowStride = b.strides()[0];
45 const std::size_t bColStride = b.strides()[1];
46
47 Tensor<T> result({m, n});
48 T* out = result.data();
49 const std::size_t outRowStride = result.strides()[0];
50 const std::size_t outColStride = result.strides()[1];
51
52 for (std::size_t i = 0; i < m; ++i) {
53 const T* aRow = aData + (i * aRowStride);
54 T* outRow = out + (i * outRowStride);
55 for (std::size_t j = 0; j < n; ++j) {
56 T total{};
57 for (std::size_t p = 0; p < k; ++p) {
58 total = static_cast<T>(total + aRow[p * aColStride] *
59 bData[(p * bRowStride) + (j * bColStride)]);
60 }
61 outRow[j * outColStride] = total;
62 }
63 }
64 return result;
65}
