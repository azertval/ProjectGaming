28template <typename T, typename Fn>
29void forEachOffset(const Tensor<T>& a, Fn&& fn) {
30 const std::size_t count = a.size();
31 // Chemin rapide : sur un tenseur contigu, l'offset de l'element de rang `n` EST `n`. Le
32 // parcours general ci-dessous refait sinon, pour chaque element, une somme sur les axes en
33 // relisant `strides()` -- un `std::vector` du tas -- alors que le resultat est connu d'avance.
34 // Meme ordre de visite dans les deux cas, donc meme suite d'appels a `fn`.
35 if (a.isContiguous()) {
36 for (std::size_t linear = 0; linear < count; ++linear) {
37 fn(linear);
38 }
39 return;
40 }
41
42 const std::size_t rank = a.rank();
43 const std::vector<std::size_t>& strides = a.strides();
44 const std::vector<std::size_t>& shape = a.shape();
45 std::vector<std::size_t> index(rank, 0);
46 for (std::size_t linear = 0; linear < count; ++linear) {
47 std::size_t offset = 0;
48 for (std::size_t axis = 0; axis < rank; ++axis) {
49 offset += index[axis] * strides[axis];
50 }
51 fn(offset);
52 for (std::size_t axis = rank; axis-- > 0;) {
53 if (++index[axis] < shape[axis]) {
54 break;
55 }
56 index[axis] = 0;
57 }
58 }
59}

62template <typename T, typename BinaryOp>
63[[nodiscard]] Tensor<T> elementwise(const Tensor<T>& a, const Tensor<T>& b, BinaryOp op) {
64 PROJECTGAMING_ASSERT(a.shape() == b.shape(), "Operation elementwise : formes incompatibles");
65 Tensor<T> result(a.shape());
66 T* out = result.data();
67 const T* aData = a.data();
68 const T* bData = b.data();
69 const std::size_t count = a.size();
70
71 // Chemin rapide, cas de loin le plus frequent (tout tenseur construit par forme l'est) : les
72 // trois parcours coincident alors terme a terme. Meme ordre, donc resultat identique au bit
73 // pres -- c'est un recalcul d'adresses supprime, pas un changement de calcul.
74 if (a.isContiguous() && b.isContiguous()) {
75 for (std::size_t linear = 0; linear < count; ++linear) {
76 out[linear] = op(aData[linear], bData[linear]);
77 }
78 return result;
79 }
80
81 const std::size_t rank = a.rank();
82 const std::vector<std::size_t>& stridesA = a.strides();
83 const std::vector<std::size_t>& stridesB = b.strides();
84 const std::vector<std::size_t>& shape = a.shape();
85 std::vector<std::size_t> index(rank, 0);
86 for (std::size_t linear = 0; linear < count; ++linear) {
87 std::size_t offsetA = 0;
88 std::size_t offsetB = 0;
89 for (std::size_t axis = 0; axis < rank; ++axis) {
90 offsetA += index[axis] * stridesA[axis];
91 offsetB += index[axis] * stridesB[axis];
92 }
93 out[linear] = op(aData[offsetA], bData[offsetB]);
94 for (std::size_t axis = rank; axis-- > 0;) {
95 if (++index[axis] < shape[axis]) {
96 break;
97 }
98 index[axis] = 0;
99 }
100 }
101 return result;
102}

105template <typename T, typename UnaryOp>
106[[nodiscard]] Tensor<T> elementwiseUnary(const Tensor<T>& a, UnaryOp op) {
107 Tensor<T> result(a.shape());
108 T* out = result.data();
109 std::size_t linear = 0;
110 forEachOffset(a, [&](std::size_t offset) { out[linear++] = op(a.data()[offset]); });
111 return result;
112}

221template <typename T>
222[[nodiscard]] T max(const Tensor<T>& a) {
223 PROJECTGAMING_ASSERT(a.size() > 0, "max() : tenseur vide");
224 bool first = true;
225 T best{};
226 detail::forEachOffset(a, [&](std::size_t offset) {
227 const T value = a.data()[offset];
228 if (first || value > best) {
229 best = value;
230 first = false;
231 }
232 });
233 return best;
234}
