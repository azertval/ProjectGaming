26void forEachOffset(const Tensor<T>& a, Fn&& fn) {
27 const std::size_t rank = a.rank();
28 std::vector<std::size_t> index(rank, 0);
29 for (std::size_t linear = 0; linear < a.size(); ++linear) {
30 std::size_t offset = 0;

32 offset += index[axis] * a.strides()[axis];
33 }
34 fn(offset);

36 if (++index[axis] < a.shape()[axis]) {

56 offsetA += index[axis] * a.strides()[axis];

71template <typename T, typename UnaryOp>
72[[nodiscard]] Tensor<T> elementwiseUnary(const Tensor<T>& a, UnaryOp op) {
73 Tensor<T> result(a.shape());
74 T* out = result.data();
75 std::size_t linear = 0;
76 forEachOffset(a, [&](std::size_t offset) { out[linear++] = op(a.data()[offset]); });
77 return result;
