24[[nodiscard]] inline std::size_t shapeProduct(const std::vector<std::size_t>& shape) {
25 std::size_t product = 1;
26 for (std::size_t dimension : shape) {
27 product *= dimension;
28 }
29 return product;
30}

33[[nodiscard]] inline std::vector<std::size_t> rowMajorStrides(
34 const std::vector<std::size_t>& shape) {
35 std::vector<std::size_t> strides(shape.size());
36 std::size_t stride = 1;
37 for (std::size_t axis = shape.size(); axis-- > 0;) {
38 strides[axis] = stride;
39 stride *= shape[axis];
40 }
41 return strides;
42}

172 [[nodiscard]] std::size_t offsetOf(std::initializer_list<std::size_t> indices) const {
173 PROJECTGAMING_ASSERT(indices.size() == rank(),
174 "Tensor::at() : nombre d'indices different du rang du tenseur");
175 std::size_t offset = 0;
176 std::size_t axis = 0;
177 for (std::size_t index : indices) {
178 PROJECTGAMING_ASSERT(index < _shape[axis], "Tensor::at() : indice hors bornes");
179 offset += index * _strides[axis];
180 ++axis;
181 }
182 return offset;
183 }
