86[[nodiscard]] inline GradientCheckResult checkGradient(
87 const std::function<autodiff::NodePtr(const std::vector<autodiff::NodePtr>&)>& buildGraph,
88 const std::vector<Tensor<float>>& inputs, float epsilon = 1e-3f, float tolerance = 1e-2f) {
89 std::vector<autodiff::NodePtr> analyticNodes;
90 analyticNodes.reserve(inputs.size());
91 for (const Tensor<float>& input : inputs) {
92 analyticNodes.push_back(autodiff::variable(input.clone()));
93 }
94 autodiff::backward(detail::reduceToScalar(buildGraph(analyticNodes)));
95
96 float maxAbsoluteError = 0.0f;
97
98 for (std::size_t inputIndex = 0; inputIndex < inputs.size(); ++inputIndex) {
99 const Tensor<float>& original = inputs[inputIndex];
100 for (std::size_t elementIndex = 0; elementIndex < original.size(); ++elementIndex) {
101 Tensor<float> perturbedPlus = original.clone();
102 perturbedPlus.data()[elementIndex] += epsilon;
103 Tensor<float> perturbedMinus = original.clone();
104 perturbedMinus.data()[elementIndex] -= epsilon;
105
106 std::vector<autodiff::NodePtr> nodesPlus;
107 std::vector<autodiff::NodePtr> nodesMinus;
108 nodesPlus.reserve(inputs.size());
109 nodesMinus.reserve(inputs.size());
110 for (std::size_t i = 0; i < inputs.size(); ++i) {
111 nodesPlus.push_back(
112 autodiff::variable(i == inputIndex ? perturbedPlus : inputs[i].clone()));
113 nodesMinus.push_back(
114 autodiff::variable(i == inputIndex ? perturbedMinus : inputs[i].clone()));
115 }
116
117 const float outputPlus = detail::reduceToScalar(buildGraph(nodesPlus))->value.at({0});
118 const float outputMinus = detail::reduceToScalar(buildGraph(nodesMinus))->value.at({0});
119 const float numericGrad = (outputPlus - outputMinus) / (2.0f * epsilon);
120
121 const float analyticGrad = analyticNodes[inputIndex]->grad.data()[elementIndex];
122
123 const float absoluteError = std::abs(numericGrad - analyticGrad);
124 if (absoluteError > maxAbsoluteError) {
125 maxAbsoluteError = absoluteError;
126 }
127 }
128 }
129
130 return GradientCheckResult{maxAbsoluteError < tolerance, maxAbsoluteError};
131}
