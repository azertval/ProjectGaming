116 [[nodiscard]] int skinIndexOf(std::string_view asset, core::TileType type) const noexcept {
117 const bool masked = hasSilhouette(type);
118 for (std::size_t index = 0; index < skins.size(); ++index) {
119 const SkinTexture& skin = skins[index];
120 if (skin.asset != asset) {
121 continue;
122 }
123 if (masked ? (skin.maskType == type) : !skin.maskType.has_value()) {
124 return static_cast<int>(index);
125 }
126 }
127 return -1;
128 }
