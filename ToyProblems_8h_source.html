39struct PolynomialToyProblem {
40 static constexpr float COEFF_A = 2.0f;
41 static constexpr float COEFF_B = -3.0f;
42 static constexpr float COEFF_C = 1.0f;
43 static constexpr std::size_t SAMPLE_COUNT = 20;
44
45 struct Sample {
46 float x;
47 float y;
48 };
49
52 [[nodiscard]] static std::vector<Sample> generateSamples(Rng& rng) {
53 std::vector<Sample> samples;
54 samples.reserve(SAMPLE_COUNT);
55 for (std::size_t i = 0; i < SAMPLE_COUNT; ++i) {
56 const float x = rng.nextFloat(-3.0f, 3.0f);
57 const float y = COEFF_A * x * x + COEFF_B * x + COEFF_C;
58 samples.push_back(Sample{x, y});
59 }
60 return samples;
61 }
62
66 [[nodiscard]] static autodiff::NodePtr loss(const autodiff::NodePtr& weights,
67 const std::vector<Sample>& samples) {
68 autodiff::NodePtr totalSquaredError;
69 for (const Sample& sample : samples) {
70 Tensor<float> featureData({3, 1});
71 featureData.at({0, 0}) = sample.x * sample.x;
72 featureData.at({1, 0}) = sample.x;
73 featureData.at({2, 0}) = 1.0f;
74 const autodiff::NodePtr features = autodiff::variable(featureData);
75
76 const autodiff::NodePtr prediction = autodiff::matmul(weights, features);
77 const autodiff::NodePtr error = autodiff::addScalar(prediction, -sample.y);
78 const autodiff::NodePtr squaredError = autodiff::multiply(error, error);
79
80 totalSquaredError =
81 totalSquaredError ? autodiff::add(totalSquaredError, squaredError) : squaredError;
82 }
83 return autodiff::multiplyScalar(totalSquaredError,
84 1.0f / static_cast<float>(samples.size()));
85 }
86};
