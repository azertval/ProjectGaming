44// Vrai quand le personnage approche le bord droit @p edge (en cases) d'une plateforme, dans une
45// fenetre de @p window cases. Declencher un saut ou une ruee AVANT cette fenetre gaspille la portee
46// horizontale sur du sol encore solide -- constat deja fait au LOT-65 TACHE-02 sur demo-budget, et
47// qui vaut pour toute fosse.
48bool atLedge(float x, float edge, float window) {
49 return x >= edge - window && x <= edge - 0.05f;
50}
51
52// Un danger temporise est-il mortel au pas @p step ? Meme formule que
53// `core::DangerController::isBlinkActive`. Un tableau qui pose un danger temporise SUR le chemin
54// exige d'attendre sa fenetre inoffensive, ce qu'aucune entree reactive ne peut deviner depuis la
55// seule position du personnage : le script doit pouvoir le calculer.
56bool blinkActive(int step, int phase, int period, int activeDuration) {
57 const int cursor = (((step - phase) % period) + period) % period;
58 return cursor < activeDuration;
59}

61// Distance parcourue par une plateforme mobile le long de son segment au pas @p step, meme formule
62// que `core::PlatformController::boxAtStep` (aller-retour triangulaire 0 -> distance -> 0). Meme
63// raison que ci-dessus : attendre qu'une plateforme revienne ne se deduit pas de la position du
64// personnage.
65float platformOffset(int step, float speed, float distance) {
66 const float travelled = static_cast<float>(step) * speed / 60.0f;
67 const float cycle = distance * 2.0f;
68 float phase = std::fmod(travelled, cycle);
69 if (phase < 0.0f) {
70 phase += cycle;
71 }
72 return phase <= distance ? phase : cycle - phase;
73}

319 // Attendre devant chaque danger temporise que sa fenetre mortelle soit passee, avec
320 // assez de marge pour traverser la case avant qu'elle ne revienne.
321 constexpr int PERIOD = 180;
322 constexpr int ACTIVE = 45;
323 constexpr int CROSSING = 40; // pas necessaires pour degager la case
324 const int phases[3] = {0, 60, 120};
325 const float holds[3] = {13.4f, 19.4f, 25.4f};
326 for (int index = 0; index < 3; ++index) {
327 if (x < holds[index] || x > holds[index] + 0.2f) {
328 continue;
329 }
330 if (blinkActive(step, phases[index], PERIOD, ACTIVE) ||
331 blinkActive(step + CROSSING, phases[index], PERIOD, ACTIVE)) {
332 in.moveX = 0.0f; // patienter : la case est (ou redevient) mortelle
333 }
